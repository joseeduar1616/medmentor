rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Os dados de estudo: cada pessoa lê e escreve só os seus.
    match /usuarios/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    // Vínculo entre e-mail e conta, usado pela função de compra para
    // descobrir quem pagou. A pessoa só pode gravar o próprio e-mail.
    match /emails/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if request.auth != null
                   && request.auth.uid == uid
                   && request.resource.data.email == request.auth.token.email.lower();
    }

    // A assinatura é somente leitura para quem está logado.
    // Quem grava é a função de compra, no servidor, com a conta de
    // serviço, que passa por cima destas regras. Sem isso qualquer pessoa
    // se daria acesso pago pelo console do navegador.
    match /assinaturas/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false;
    }

    // Números que aparecem no ranking das salas de amigos: nome, minutos
    // estudados, questões e acertos. Cada pessoa escreve só o seu.
    //
    // A leitura também é só do próprio dono. Quem monta o ranking é o
    // servidor, com a conta de serviço: sem isso bastaria saber o uid de
    // alguém para ler os números dessa pessoa sem estar em sala nenhuma.
    match /perfis/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    // Salas de amigos. Fechadas para o navegador nas duas pontas: o
    // documento guarda o PBKDF2 da senha, e quem lê ou escreve é a rota
    // /api/salas, no servidor, depois de conferir a senha.
    match /salas/{sala} {
      allow read, write: if false;
    }

    // Os recados da sala ficam pendurados nela, em salas/{sala}/mensagens.
    // A regra de cima não alcança subcoleção, e o que não tem regra fica
    // sem permissão nenhuma — mas escrever isto aqui deixa a intenção no
    // papel e não depende de o Firestore continuar assim.
    match /salas/{sala}/{resto=**} {
      allow read, write: if false;
    }

    // Baralhos que o dono publica. Também fechados para o navegador: quem
    // publica é só o dono e quem baixa é só quem tem plano em dia, e as
    // duas coisas são conferidas na rota /api/baralhos. Liberar a leitura
    // aqui entregaria o conteúdo pago a qualquer pessoa logada.
    match /publicos/{baralho} {
      allow read, write: if false;
    }

    // Provas comentadas que o dono publica com um código. Fechadas para o
    // navegador: o código É a chave, e quem pudesse listar a coleção teria
    // todas as provas sem código nenhum. Quem lê é a rota /api/provas, uma
    // prova por vez, pelo código exato. As questões ficam em
    // provas/{codigo}/questoes/{n}, porque uma prova comentada inteira
    // passaria do limite de 1 MiB de um documento só.
    match /provas/{codigo} {
      allow read, write: if false;
    }
    match /provas/{codigo}/{resto=**} {
      allow read, write: if false;
    }

    // A ligação com o Notion de cada pessoa. Guarda o token de leitura do
    // Notion dela, então é a coleção mais fechada de todas: nem o próprio
    // dono lê daqui. Quem usa é a rota /api/notion, no servidor, e ela
    // nunca devolve o token numa resposta.
    match /notion/{uid} {
      allow read, write: if false;
    }

    // Os aparelhos que assinaram a notificação por push, em
    // push/{uid}/aparelhos/{id}. Fechados para o navegador nas duas
    // pontas: cada registro guarda o endereço de entrega e a chave do
    // aparelho, e quem tivesse a chave de outra pessoa conseguiria mandar
    // notificação em nome do site direto para o celular dela. Quem lê e
    // grava é a rota /api/push, com a conta de serviço.
    match /push/{uid} {
      allow read, write: if false;
    }
    match /push/{uid}/{resto=**} {
      allow read, write: if false;
    }

    // A chave do site para assinar as notificações (VAPID), gerada uma vez
    // pelo próprio servidor. A parte privada está aqui dentro: quem a
    // lesse poderia mandar notificação em nome do Cadência para qualquer
    // aparelho assinado.
    match /config/{item} {
      allow read, write: if false;
    }

    // Quanto cada pessoa gastou de IA hoje. Fechado para o navegador pelo
    // motivo óbvio: quem pudesse escrever aqui zeraria o próprio contador
    // e o teto deixaria de existir. Quem conta é o servidor, em _ia.js.
    match /uso/{uid} {
      allow read, write: if false;
    }

    // O histórico do assistente, em conversas/{uid}/itens/{id}. Fechado
    // para o navegador: quem lê e grava é a rota /api/conversas, que tira
    // o uid do token conferido. Uma conversa com o assistente pode conter
    // o que a pessoa não contaria a ninguém; ela não pode ficar ao alcance
    // de quem souber o uid de outra.
    match /conversas/{uid}/{resto=**} {
      allow read, write: if false;
    }

    // Papel de mentor e a lista de alunos de cada mentor (por uid, vindo
    // de emails/{uid}). Só o servidor mexe aqui: /api/cupom concede o
    // papel ao resgatar "mentor1612", e /api/mentor adiciona, remove e lê
    // a lista — as duas com a conta de serviço.
    match /mentores/{uid} {
      allow read, write: if false;
    }
  }
}
